    <script>
        let currentStep = 1;
        let selectedDate = "";
        let picksLeft = 3;
        let chosenCards = [];

        const cardOptions = [
            "🍰 Гоё газар амттай бялуу идэх",
            "☕ Тухтай кафе орж кофе уух",
            "📸 Хамтдаа гоё зураг татуулах",
            "🎬 Гоё кино үзэх vibe-тай байх",
            "🍜 Амттай хоол зооглох",
            "🎁 Бэлгээ задлах мөч",
            "🌙 Орой хотын өнгө үзэх",
            "🎶 Хөгжим сонсон алхах",
            "✨ Чиллэх гоё газар олох"
        ];

        function nextStep(step) {
            document.getElementById(`step-${currentStep}`).classList.remove('active');
            currentStep = step;
            document.getElementById(`step-${currentStep}`).classList.add('active');
            if (currentStep === 3) { initMysteryCards(); }
        }

        function validateDate() {
            const dateInput = document.getElementById('selected-date').value;
            if (!dateInput) {
                alert("Эхлээд төрсөн өдрийн өдрөө сонгоно уу!");
                return;
            }
            selectedDate = dateInput;
            document.getElementById('final-date-text').innerText = selectedDate;
            nextStep(3);
        }

        function initMysteryCards() {
            const grid = document.getElementById('cards-grid');
            grid.innerHTML = "";
            cardOptions.forEach((text, index) => {
                const card = document.createElement('div');
                card.className = 'mystery-card';
                card.innerText = `Card #${index + 1}`;
                card.onclick = () => revealCard(card, text);
                grid.appendChild(card);
            });
        }

        function revealCard(element, text) {
            if (element.classList.contains('revealed') || picksLeft <= 0) return;
            element.classList.add('revealed');
            element.innerText = text;
            chosenCards.push(text);
            picksLeft--;
            document.getElementById('picks-left').innerText = picksLeft;
            if (picksLeft === 0) {
                document.querySelectorAll('.mystery-card:not(.revealed)').forEach(c => {
                    c.style.opacity = '0.4';
                    c.style.pointerEvents = 'none';
                });
                document.getElementById('to-final-btn').style.display = 'block';
                const list = document.getElementById('final-choices-list');
                list.innerHTML = "";
                chosenCards.forEach(item => {
                    let li = document.createElement('li');
                    li.innerText = item;
                    list.appendChild(li);
                });
            }
        }

        function finishInvite() {
            nextStep(5);
        }
    </script>
